    function selectOperation(type) {
        document.querySelectorAll('.op-btn').forEach(b => b.classList.remove('active'));

        if (activeBtn) activeBtn.classList.add('active');

    function createRandomPower() {
        const exp = Math.floor(Math.random() * 7) + 2; 
        let maxBase = 10;
        if (exp === 3) maxBase = 9;
        else if (exp === 4) maxBase = 6;
        else if (exp === 5) maxBase = 4;
        else if (exp >= 6) maxBase = 3; 
        
        const base = Math.floor(Math.random() * (maxBase - 2 + 1)) + 2;
        return { base: base, exp: exp, val: Math.pow(base, exp) };
    }

    function generateExample() {

        } 
        else if (currentOperation === 'sub') {
            num1 = Math.floor(Math.random() * 900000) + 10000;
            num2 = Math.floor(Math.random() * (num1 - 1000)) + 500;
            currentCorrectAnswer = num1 - num2;

            num1 = Math.floor(Math.random() * 2500) + 120;
            num2 = Math.floor(Math.random() * 450) + 12;
            currentCorrectAnswer = num1 * num2;

            num2 = Math.floor(Math.random() * 350) + 12;
            currentCorrectAnswer = Math.floor(Math.random() * 4500) + 105;
            num1 = num2 * currentCorrectAnswer;
        }
        else if (currentOperation === 'pow') {
            const isSum = Math.random() < 0.5; 
            const p1 = createRandomPower();
            if (isSum) {
                const p2 = createRandomPower();
                powData = { isSum: true, p1: p1, p2: p2 };
                currentCorrectAnswer = p1.val + p2.val;
            } else {
                powData = { isSum: false, p1: p1 };
                currentCorrectAnswer = p1.val;
            }
        }
        else if (currentOperation === 'geom') {
            let idx;
            do {
                idx = Math.floor(Math.random() * geometryTasks.length);
            } while (idx === lastGeomIndex && geometryTasks.length > 1);
            lastGeomIndex = idx;
            currentGeomTask = geometryTasks.at(idx);
        }
        
        buildInteractiveGrid();
    }

            container.appendChild(box);
            return;

            let ansInputs = [];
            for (let i = 0; i < ansStr.length; i++) {
                let cell = createCell('', true, ansStr.charAt(i));
                cell.style.borderBottom = "2px solid #2c3e50";
                mainRow.appendChild(cell);
                let inp = cell.querySelector('input');
                solutionMatrix.push(inp);
                ansInputs.push(inp);
                colIdx++;
            }

            inputFlow.push(...ansInputs);

            while (colIdx < totalCols) {
                mainRow.appendChild(createCell(''));
                colIdx++;
            }
            container.appendChild(mainRow);

            let botPadRow = createRowElement();
            for (let i = 0; i < totalCols; i++) botPadRow.appendChild(createCell(''));
            container.appendChild(botPadRow);

            for (let r = 0; r < 5; r++) {
                let scratchRow = createRowElement();
                let rowInputs = [];
                for (let c = 0; c < totalCols; c++) {
                    if (c >= 2 && c < totalCols - 1) {
                        let cell = createCell('', true, ' ');
                        scratchRow.appendChild(cell);
                        rowInputs.push(cell.querySelector('input'));
                    } else {
                        scratchRow.appendChild(createCell(''));
                    }
                }
                inputFlow.push(...rowInputs);
                container.appendChild(scratchRow);
            }

            bindFlowNavigation();
            focusFirstInput(ansInputs.at(0));
            return;
        }
        let s1 = num1.toString();
        let s2 = num2.toString();
        let sign = currentOperation === 'add' ? '+' : currentOperation === 'sub' ? '-' : currentOperation === 'mul' ? '×' : '÷';

        if (currentOperation !== 'div') {
            let len = Math.max(s1.length, s2.length, ansStr.length);
            let totalCols = len + 4;

            let row1 = createRowElement();
            let startCol1 = totalCols - s1.length;
            for (let i = 0; i < totalCols; i++) {
                row1.appendChild(createCell((i >= startCol1) ? s1.charAt(i - startCol1) : ''));
            }
            container.appendChild(row1);

            let row2 = createRowElement();
            let startCol2 = totalCols - s2.length;
            for (let i = 0; i < totalCols; i++) {
                let char = '';
                if (i === startCol2 - 2) char = sign;
                else if (i >= startCol2) char = s2.charAt(i - startCol2);
                row2.appendChild(createCell(char));
            }
            row2.classList.add('solid-line');
            container.appendChild(row2);

            if (currentOperation === 'mul' && s2.length > 1) {
                for (let r = 0; r < s2.length; r++) {
                    let tempRow = createRowElement();
                    let rowCells = [];
                    for (let i = 0; i < totalCols; i++) {
                        let cell = createCell('', true, ' ');
                        tempRow.appendChild(cell);
                        rowCells.push(cell.querySelector('input'));
                    }
                    if (r === s2.length - 1) tempRow.classList.add('solid-line');
                    container.appendChild(tempRow);

                    let activePart = rowCells.slice(1, totalCols - r).reverse();
                    inputFlow.push(...activePart);
                }
            }

            let rowAns = createRowElement();
            let startAnsCol = totalCols - ansStr.length;
            let ansInputs = [];
            for (let i = 0; i < totalCols; i++) {
                if (i >= startAnsCol) {
                    let expectedChar = ansStr.charAt(i - startAnsCol);
                    let cell = createCell('', true, expectedChar);
                    rowAns.appendChild(cell);
                    let inp = cell.querySelector('input');
                    solutionMatrix.push(inp);
                    ansInputs.push(inp);
                } else {
                    rowAns.appendChild(createCell(''));
                }
            }
            container.appendChild(rowAns);

            inputFlow.push(...ansInputs.slice().reverse());

            bindFlowNavigation();
            focusFirstInput(inputFlow.at(0));
        } else {
            let totalCols = Math.max(s1.length + s2.length + 4, 16);
            
            let row1 = createRowElement();
            for (let i = 0; i < totalCols; i++) {
                let cell;
                if (i < s1.length) cell = createCell(s1.charAt(i));
                else if (i === s1.length) { cell = createCell(''); cell.classList.add('div-corner-right'); }
                else if (i > s1.length && i <= s1.length + s2.length) cell = createCell(s2.charAt(i - s1.length - 1));
                else cell = createCell('');
                row1.appendChild(cell);
            }
            container.appendChild(row1);

            let row2 = createRowElement();
            let startAnsCol = s1.length + 1;
            let leftRow2Inputs = [];
            let quotientInputs = [];

            for (let i = 0; i < totalCols; i++) {
                let cell;
                if (i < s1.length) {
                    cell = createCell('', true, ' '); 
                    leftRow2Inputs.push(cell.querySelector('input'));
                } else if (i === s1.length) {
                    cell = createCell(''); cell.classList.add('div-corner-right');
                } else if (i >= startAnsCol && i < startAnsCol + ansStr.length) {
                    let expectedChar = ansStr.charAt(i - startAnsCol);
                    cell = createCell('', true, expectedChar);
                    cell.classList.add('div-corner-bottom');
                    let inp = cell.querySelector('input');
                    solutionMatrix.push(inp);
                    quotientInputs.push(inp);
                } else {
                    cell = createCell('');
                    if (i >= startAnsCol) cell.classList.add('div-corner-bottom');
                }
                row2.appendChild(cell);
            }
            container.appendChild(row2);

            inputFlow.push(...quotientInputs);
            inputFlow.push(...leftRow2Inputs);

            for (let r = 0; r < 8; r++) {
                let row = createRowElement();
                let calcInputs = [];
                for (let c = 0; c < totalCols; c++) {
                    let cell;
                    if (c <= s1.length + 2) {
                        cell = createCell('', true, ' '); 
                        calcInputs.push(cell.querySelector('input'));
                        if (r % 2 === 1 && c < s1.length + 1) {
                            cell.style.borderBottom = "2px solid #2c3e50"; 
                        }
                    } else {
                        cell = createCell('');
                    }
                    row.appendChild(cell);
                }
                inputFlow.push(...calcInputs);
                container.appendChild(row);
            }

            bindFlowNavigation();
            focusFirstInput(quotientInputs.at(0));
        }
    }
    function createRowElement() {
        let row = document.createElement('div');
        row.className = 'matrix-row';
        return row;
    }

    function createCell(char = '', isInput = false, expected = '') {
        let cell = document.createElement('div');
        cell.className = 'matrix-cell';
        if (isInput) {
            let input = document.createElement('input');
            input.type = 'text';
            input.inputMode = 'numeric';
            input.pattern = '[0-9]*';
            input.maxLength = 1;
            input.autocomplete = 'off';
            if (expected !== ' ') {
                input.dataset.expected = expected; 
            }
            cell.appendChild(input);
        } else {
            cell.textContent = char;
        }
        return cell;
    }

    function bindFlowNavigation() {
        inputFlow.forEach((inp, idx) => {
            inp.addEventListener('input', function() {
                this.value = this.value.replace(/[^0-9]/g, '');
                this.className = '';
                const nextInp = inputFlow.at(idx + 1);
                if (this.value.length === 1 && nextInp) {
                    nextInp.focus();
                }
            });

            inp.addEventListener('keydown', function(e) {
                if (e.key === 'Backspace') {
                    if (this.value === '' && idx - 1 >= 0) {
                        e.preventDefault();
                        const prevInp = inputFlow.at(idx - 1);
                        if (prevInp) {
                            prevInp.focus();
                            prevInp.value = '';
                        }
                    }
                } else if (e.key === 'Enter') {
                    e.preventDefault();
                    const allAnsFilled = solutionMatrix.every(el => el.value.trim() !== '');
                    if (allAnsFilled) {
                        checkAnswer();
                    } else if (solutionMatrix.length > 0) {
                        const firstEmptyAns = solutionMatrix.slice().reverse().find(el => el.value.trim() === '');
                        if (firstEmptyAns) firstEmptyAns.focus();
                    }
                } else if (e.key === 'ArrowLeft') {
                    let prevCell = this.parentElement.previousElementSibling;
                    if (prevCell && prevCell.querySelector('input')) prevCell.querySelector('input').focus();
                } else if (e.key === 'ArrowRight') {
                    let nextCell = this.parentElement.nextElementSibling;
                    if (nextCell && nextCell.querySelector('input')) nextCell.querySelector('input').focus();
                }
            });

            inp.addEventListener('focus', function() {
                this.select();
            });
        });
    }

    function checkAnswer() {
        const msg = document.getElementById('result-msg');

        if (currentOperation === 'geom') {
            if (!selectedGeomAnswer) {
                msg.innerHTML = "Сначала нажми на один из трёх вариантов!";
                msg.className = "incorrect-text";
                return;
            }

            const buttons = document.querySelectorAll('.geom-opt-btn');
            buttons.forEach(btn => {
                btn.disabled = true;
                if (btn.textContent === currentGeomTask.name) {
                    btn.classList.add('cell-correct');
                } else if (btn.textContent === selectedGeomAnswer && selectedGeomAnswer !== currentGeomTask.name) {
                    btn.classList.add('cell-incorrect');
                }
            });

            if (selectedGeomAnswer === currentGeomTask.name) {
                correctCount++;
                msg.innerHTML = `Верно! Это ${currentGeomTask.name.toLowerCase()}. 🌟`;
                msg.className = "correct-text";
            } else {
                msg.innerHTML = `Ошибка! Правильный ответ: ${currentGeomTask.name}.`;
                msg.className = "incorrect-text";
            }

            document.getElementById('check-btn').style.display = 'none';
            document.getElementById('next-btn').style.display = 'flex';
            document.getElementById('correct-counter').textContent = correctCount;
            return;
        }

        let allCorrect = true;
        let hasEmptyFields = false;

        solutionMatrix.forEach(input => {
            if (input.value.trim() === "") {
                hasEmptyFields = true;
            }
        });

        if (hasEmptyFields) {
            msg.innerHTML = "Сначала заполни все клетки ответа!";
            msg.className = "incorrect-text";
            const emptyCell = solutionMatrix.slice().reverse().find(el => el.value.trim() === "");
            if (emptyCell) emptyCell.focus();
            return;
        }

        document.querySelectorAll('.matrix-cell input').forEach(inp => inp.disabled = true);

        solutionMatrix.forEach(input => {
            let userVal = input.value.trim();
            let correctVal = input.dataset.expected;

            if (userVal === correctVal) {
                input.className = 'cell-correct';
            } else {
                input.className = 'cell-incorrect';
                allCorrect = false;
            }
        });

        if (allCorrect) {
            correctCount++;
            msg.innerHTML = "Абсолютно верно! Присвоено 1 очко. 🌟";
            msg.className = "correct-text";
        } else {
            msg.innerHTML = `Ошибка! Правильный ответ: ${currentCorrectAnswer}.`;
            msg.className = "incorrect-text";
        }

        document.getElementById('check-btn').style.display = 'none';
        document.getElementById('next-btn').style.display = 'flex';
        document.getElementById('correct-counter').textContent = correctCount;
    }

    const ctx = canvas.getContext('2d');
    const pad = document.getElementById('pad');
    const padHeader = document.getElementById('pad-header');
    const btnPadSize = document.getElementById('btn-pad-size');
    const btnPadClear = document.getElementById('btn-pad-clear');
    let activeDrawPointer = null;

    function resizeCanvas() {
        if (pad.style.display !== 'flex') return;
        const w = pad.clientWidth;
        const h = pad.clientHeight - padHeader.offsetHeight;
        if (w <= 0 || h <= 0) return;

        const tempCanvas = document.createElement('canvas');
        tempCanvas.width = canvas.width;
        tempCanvas.height = canvas.height;
        const tempCtx = tempCanvas.getContext('2d');
        if (canvas.width > 0 && canvas.height > 0) {
            tempCtx.drawImage(canvas, 0, 0);
        }
        
        canvas.width = w;
        canvas.height = h;
        
        ctx.strokeStyle = '#0052cc';
        ctx.lineWidth = 3;
        ctx.lineCap = 'round';
        ctx.lineJoin = 'round';
        if (tempCanvas.width > 0 && tempCanvas.height > 0) {
            ctx.drawImage(tempCanvas, 0, 0);
        }
    }

    btnPadSize.addEventListener('click', function(e) {
        e.stopPropagation();
        pad.classList.toggle('expanded');
        requestAnimationFrame(resizeCanvas);
    });

    btnPadClear.addEventListener('click', function(e) {
        e.stopPropagation();
        ctx.clearRect(0, 0, canvas.width, canvas.height);
    });

    canvas.addEventListener('pointerdown', function(e) {
        if (activeDrawPointer !== null) return;
        activeDrawPointer = e.pointerId;
        canvas.setPointerCapture(e.pointerId);
        const rect = canvas.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;
        ctx.beginPath();
        ctx.moveTo(x, y);
        ctx.lineTo(x + 0.1, y + 0.1);
        ctx.stroke();
    });

    canvas.addEventListener('pointermove', function(e) {
        if (e.pointerId !== activeDrawPointer) return;
        const rect = canvas.getBoundingClientRect();
        const x = e.clientX - rect.left;
        const y = e.clientY - rect.top;
        ctx.lineTo(x, y);
        ctx.stroke();
    });

    function endDrawPointer(e) {
        if (e.pointerId === activeDrawPointer) {
            activeDrawPointer = null;
        }
    }
    canvas.addEventListener('pointerup', endDrawPointer);
    canvas.addEventListener('pointercancel', endDrawPointer);

    let activeDragPointer = null;
    let dragOffsetX = 0;
    let dragOffsetY = 0;

    padHeader.addEventListener('pointerdown', function(e) {
        if (e.target.tagName === 'BUTTON') return;
        activeDragPointer = e.pointerId;
        padHeader.setPointerCapture(e.pointerId);
        const rect = pad.getBoundingClientRect();
        dragOffsetX = e.clientX - rect.left;
        dragOffsetY = e.clientY - rect.top;
        pad.style.bottom = 'auto';
        pad.style.right = 'auto';
        pad.style.left = rect.left + 'px';
        pad.style.top = rect.top + 'px';
    });

    padHeader.addEventListener('pointermove', function(e) {
        if (e.pointerId !== activeDragPointer) return;
        const newLeft = Math.max(0, Math.min(window.innerWidth - pad.offsetWidth, e.clientX - dragOffsetX));
        const newTop = Math.max(0, Math.min(window.innerHeight - pad.offsetHeight, e.clientY - dragOffsetY));
        pad.style.left = newLeft + 'px';
        pad.style.top = newTop + 'px';
    });

    function endDragPointer(e) {
        if (e.pointerId === activeDragPointer) {
            activeDragPointer = null;
        }
    }
    padHeader.addEventListener('pointerup', endDragPointer);
    padHeader.addEventListener('pointercancel', endDragPointer);
